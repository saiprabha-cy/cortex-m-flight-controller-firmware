# Host-side unit tests (native gcc). Build:  cmake -S tests/unit -B build-host && cmake --build build-host && ctest --test-dir build-host
cmake_minimum_required(VERSION 3.16)
project(fc_host_tests C)
set(CMAKE_C_STANDARD 11)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
option(SANITIZE "Build with ASan+UBSan" ON)
option(COVERAGE "Build with gcov coverage" OFF)

set(CMN ${ROOT}/common)
set(MONO ${ROOT}/third_party/monocypher/monocypher.c ${ROOT}/third_party/monocypher/monocypher-ed25519.c)
add_library(unity STATIC ${ROOT}/third_party/unity/unity.c)
target_include_directories(unity PUBLIC ${ROOT}/third_party/unity)
add_library(fc_common STATIC ${CMN}/crc.c ${CMN}/sha256.c ${CMN}/frame.c ${CMN}/flashlog.c ${CMN}/image_verify.c
                              ${CMN}/boot_meta.c ${CMN}/mode_fsm.c ${CMN}/wdg_mgr.c ${CMN}/fault_policy.c ${MONO})
add_library(fc_logic STATIC ${ROOT}/app/src/kalman.c ${ROOT}/app/src/sim_sensors.c ${ROOT}/drivers/imu_mpu6500.c ${ROOT}/drivers/baro_bmp280.c)
target_include_directories(fc_logic PUBLIC ${ROOT}/app/inc ${ROOT}/drivers)
add_library(flash_model STATIC flash_model.c image_builder.c)
target_link_libraries(flash_model fc_common)

set(OPTS -Wall -Wextra -Wno-unused-parameter -g -O1)
if(SANITIZE)
  list(APPEND OPTS -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(COVERAGE)
  list(APPEND OPTS --coverage)
endif()
foreach(t fc_common fc_logic flash_model unity)
  target_compile_options(${t} PRIVATE ${OPTS})
  target_link_options(${t} PUBLIC ${OPTS})
endforeach()

enable_testing()
foreach(t crc_sha frame mode_fsm flashlog image_verify boot_meta kalman sensors wdg_policy pipeline)
  add_executable(test_${t} test_${t}.c)
  target_compile_options(test_${t} PRIVATE ${OPTS})
  target_link_options(test_${t} PRIVATE ${OPTS})
  target_link_libraries(test_${t} unity fc_common fc_logic flash_model m)
  add_test(NAME ${t} COMMAND test_${t})
endforeach()
add_executable(verify_cli verify_cli.c)
target_link_libraries(verify_cli fc_common)
target_compile_options(verify_cli PRIVATE ${OPTS})
target_link_options(verify_cli PRIVATE ${OPTS})
